지금까지 논의하며 발전시켜 온 [소득세 및 부가가치세 통합 정산 시스템]의 전체 슈도코드(Pseudocode)입니다.

누락된 부분 없이 모든 모듈(조기 반환 패턴, f-접두어, 부가세 전처리, 지방세 분리 등)을 하나의 파일처럼 복사하여 붙여넣을 수 있도록 하나로 통합했습니다.

```c
// ============================================================================
// 소득세 및 부가가치세 통합 정산 시스템 (v1.0)
// ============================================================================

// ----------------------------------------------------------------------------
// [모듈 1] 부가가치세 전처리 (데이터 정제)
// ----------------------------------------------------------------------------
func f부가세_전처리_프로세스(원천_매출_데이터) {
    
    // 최종 소비자에게 받은 금액(공급대가)에서 순수 매출(공급가액)과 부가세를 분리
    // 예: 11,000원을 받았다면 10,000원(매출)과 1,000원(부채)으로 나눔
    순수_매출액 = f공급가액_산출(원천_매출_데이터.총금액);
    납부할_부가세 = 순수_매출액 * 0.1;

    // 분리된 순수 매출액만 소득세 계산기로 전달
    return 순수_매출액;
}

// ----------------------------------------------------------------------------
// [모듈 2] 개인 소득세 메인 엔진
// ----------------------------------------------------------------------------
func f세금_계산_프로세스(주체_정보, 매출_데이터) {

    // [단계 1] 소득금액 산출 (매출액 - 경비)
    소득금액 = 매출_데이터.매출액 - f경비_인정_프로세스(주체_정보, 매출_데이터);

    // [단계 2] 과세표준 결정 (소득금액 - 소득공제)
    과세표준 = 소득금액 - f공제_항목_계산(주체_정보);

    // 가드 클로즈: 세금을 매길 기초 금액이 없으면 즉시 종료 (결손금)
    if (과세표준 <= 0) {
        return 0;
    }

    // [단계 3] 산출세액 결정 (누진세율 적용)
    산출세액 = f세율_적용_엔진(과세표준);

    // [단계 4] 결정세액 확정 (산출세액 - 세액공제)
    세액공제_합계 = f세액공제_계산_프로세스(주체_정보);
    결정세액 = 산출세액 - 세액공제_합계;
    
    // 가드 클로즈: 공제액이 커서 낼 국세(소득세)가 없으면 즉시 종료
    if (결정세액 <= 0) {
        return 0;
    }

    // [단계 5] 국세와 지방세를 합산하여 최종 납부액 산출
    최종_납부액 = f지방소득세_합산_프로세스(결정세액);

    return 최종_납부액;
}

// ----------------------------------------------------------------------------
// [모듈 3] 경비 및 공제 세부 로직 (블랙박스)
// ----------------------------------------------------------------------------
func f경비_인정_프로세스(주체_정보, 매출_데이터) {

    // 조건 1: 근로자인 경우 (국가 정률 공식 강제 적용, 개별 영수증 무시)
    if (주체_정보.타입 == "근로자") {
        return f근로소득공제_정률_계산(매출_데이터.매출액);
    }

    // 조건 2: 사업자인 경우 (실제 증빙 기반)
    실제_비용 = 0;
    for (각_지출_항목 in 매출_데이터.지출_내역) {
        
        // 업무 연관성이 없는 지출은 스킵
        if (f업무_연관성_검증(각_지출_항목) == false) {
            continue;
        }
        실제_비용 += 각_지출_항목.금액;
    }
    
    return 실제_비용;
}

func f세액공제_계산_프로세스(주체_정보) {
    합계액 = 0;

    // 조건 1: 자녀가 있는 경우
    if (주체_정보.자녀_수 > 0) {
        합계액 += f자녀_세액공제_계산(주체_정보.자녀_수);
    }

    // 조건 2: 월세를 지불 중인 경우 (연봉 조건 포함)
    if (주체_정보.월세_지출_액 > 0) {
        합계액 += f월세_세액공제_계산(주체_정보.월세_지출_액, 주체_정보.연봉_수준);
    }

    // 조건 3: 보장성 보험료 등을 낸 경우
    if (주체_정보.보험료_지출_액 > 0) {
        합계액 += f보험료_세액공제_계산(주체_정보.보험료_지출_액);
    }

    return 합계액;
}

// ----------------------------------------------------------------------------
// [모듈 4] 지방소득세 명시적 합산 로직
// ----------------------------------------------------------------------------
func f지방소득세_합산_프로세스(국세_금액) {
    
    // 지방소득세는 확정된 국세(소득세)의 10%로 정의됨
    지방소득세 = 국세_금액 * 0.1;
    
    // 최종적으로 납세자가 고지서에 받게 될 총합계 금액
    총_납부액 = 국세_금액 + 지방소득세;

    return 총_납부액;
}

// ----------------------------------------------------------------------------
// [모듈 5] 부가가치세 정산 메인 로직
// ----------------------------------------------------------------------------
func f부가세_정산_프로세스(매출_내역, 매입_내역) {
    
    // 매출 부가세 (매출세액): 고객에게 받아서 보관 중인 돈 (부채)
    매출_부가세 = 0;
    for (각_매출 in 매출_내역) {
        매출_부가세 += 각_매출.총금액 * (0.1 / 1.1);
    }

    // 매입 부가세 (매입세액): 물건을 살 때 내가 이미 낸 부가세 (자산/환급대상)
    매입_부가세 = 0;
    for (각_지출 in 매입_내역) {
        // 부가세 환급이 불가능한 항목(면세, 간이배제 등)은 스킵
        if (f부가세_환급_적격_검증(각_지출) == false) {
            continue;
        }
        매입_부가세 += 각_지출.금액 * (0.1 / 1.1);
    }

    // 최종 납부액 = (받아둔 돈) - (내가 미리 낸 돈)
    최종_부가세_납부액 = 매출_부가세 - 매입_부가세;

    // 내가 낸 부가세가 더 많으면 국가에서 환급(돌려줌)
    if (최종_부가세_납부액 < 0) {
        return f부가세_환급_처리(최종_부가세_납부액); 
    }

    return 최종_부가세_납부액;
}
```
